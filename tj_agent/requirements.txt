# tj_agent dependencies
#
# The base sync daemon (push/pull/sysconfig) uses only the Python stdlib
# (urllib + sqlite3). Install these only if you intend to run the optional
# remote inference worker (worker_enabled=true in config), which acts as
# an MCP client and runs an agentic loop against local ollama:
#
#     pip install -r tj_agent/requirements.txt

# MCP client SDK — talks to lxr-mcp-server, github-mcp-server, etc.
mcp>=1.27.0

# Official first-party URL fetcher MCP server (mcp-server-fetch).
# Wraps httpx + readabilipy + markdownify + protego (robots.txt) and
# is run as a stdio subprocess by the worker via `python -m mcp_server_fetch`.
mcp-server-fetch>=2025.4

# HTTP client for the bundled lxr-mcp-server, the local google_cse MCP
# server in tj_agent/mcp_servers/, and any other Python MCP servers we
# wire in-process.
httpx>=0.28
beautifulsoup4>=4.14
